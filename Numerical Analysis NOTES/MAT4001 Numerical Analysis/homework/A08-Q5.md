import matplotlib.pyplot as plt
import numpy as np

global zero

def sign(num):
    if num>0:
        return 1.0
    elif num<0:
        return -1.0
    else:
        print("cannot identify the sign of 0")

def P0(x):
    return np.ones_like(x)

def P1(x):
    return x

def P2(x):
    return (x**2) - 1/3

def P3(x):
    return (x**3) - (3/5) * x

def P4(x):
    return x**4 - (6/7)*(x**2) + 3/35

def Müller(k=2, p_last_1_k = 1, p_last_2_k = -0.9, p_last_3_k = -1):
