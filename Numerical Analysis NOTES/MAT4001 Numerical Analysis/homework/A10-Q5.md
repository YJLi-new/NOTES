import numpy as np

Tj = np.array([
    [00, -0.25, 0.25, -0.25],
    [-0.25, 0.0, 0.25, 0.25],
    [0.2, 0.2, 0.0, -0.2],
    [-0.33333333, 0.33333333, -0.33333333, 0.0]
])
cj = np.array([-0.5, -0.25, 0.0, -0.33333333])

Tg = np.array([
    [0.0, -0.25, 0.25, -0.25],
    [0.0, 0.0625, 0.1875, 0.3125],
    [0.0, -0.0375, 0.0875, -0.1875],
    [0.0, 0.1375, 0.0125, 0.35416667]
])
cg = np.array([-0.5, -0.125, -0.125, 0.45833333])
dep_j = 0
dep_g = 0

def jacobi(last_guess = np.array([0.0,0.0,0.0,0.0])):
    global Tj, cj, dep_j
    dep_j += 1
    guess = np.dot(Tj, last_guess) + cj

    res = guess - last_guess
    max_row = 0.0
    for i in range(4):
        row_sum = np.sum(np.abs(res[i]))
        if row_sum > max_row:
            max_row = row_sum

    if max_row < 0.0001:
        print("Use Jacobi method:")
        print("T = \n", Tj,
              "\n\nc = \n", cj,
              "\n\ndepth = ", dep_j,
              "\n\nresult: \n", guess)
        return guess
    else:
        jacobi(guess)

def gaussSeidel(last_guess = np.array([0.0,0.0,0.0,0.0])):
