@routes.post('/dropped')
async def handle_dropped(req: Request) -> Response:
    """Called by hub server each time a user drops an item in this domain."""
    try:
        data = await req.json()

        required_fields = ['secret', 'user', 'item']
        for field in required_fields:
            if field not in data:
                return json_response(status=400, data={'error': f'Missing field: {field}'})

        secret = data['secret']
        user_id = data['user']
        item = data['item']

        user_id = int(user_id)
        item_id = int(item['id'])

        app = req.app

        if secret != app.get('secret'):
            return json_response(status=403, data={'error': 'Invalid secret'})

        if user_id not in app['users']:
            return json_response(status=403, data={'error': f'User {user_id} not found'})

        user = app['users'][user_id]

        if item_id not in user['inventory'] or user['inventory'][item_id] != 'inventory':
            return json_response(status=403, data={'error': f'User {user_id} does not carry item {item_id}'})

        current_location = user.get('location')
        if not current_location:
            return json_response(status=400, data={'error': 'User location not set'})

        if item_id not in app['item_map']:
            return json_response(status=400, data={'error': f'Item {item_id} not recognized'})

        if current_location not in app['game_state']:
            app['game_state'][current_location] = []
        app['game_state'][current_location].append(item_id)

        user['inventory'][item_id] = current_location

        return json_response({'location': current_location})
    except Exception as e:
        print(f"Exception in /dropped: {e}")
        return json_response(status=500, data={'error': 'Internal Server Error'})

@routes.post("/command")
async def handle_command(req: Request) -> Response:
    """Handle hub-server commands"""
    try:
        data = await req.json()

        user_id = data.get("user")
        command = data.get("command")

        if user_id is None or command is None:
            return web.Response(status=400, text="Error: Missing 'user' or 'command' field.")

        if not isinstance(command, list) or not all(isinstance(word, str) for word in command):
            return web.Response(status=400, text="Error: 'command' must be a list of strings.")

        user_id = int(user_id)

        app = req.app

        user = app.get('users', {}).get(user_id)
        if user is None:
            return web.Response(status=403, text=f"Error: User {user_id} not found.")

        response = await process_command(app, user_id, command)
        return web.Response(text=response)
    except Exception as e:
        print(f"Exception in /command: {e}")
        return web.Response(status=500, text="Error: Internal Server Error")

async def process_command(app, user_id, command):
    user = app['users'][user_id]
    location = user['location']
    game_state = app['game_state']
    item_map = app['item_map']
    domain_id = app['domain_id']
    secret = app['secret']
    hub_url = app.get('hub_url')  # Retrieve the stored hub URL

    if not command:
        return "No command entered."

    verb = command[0].lower()
    args = command[1:]

    if verb == "take":
        if not args:
            return "Take what?"
        item_name = ' '.join(args).lower()
        item_id = None
        # Find the item in the current location
        for i_id in game_state.get(location, []):
            if item_map[i_id]['name'].lower() == item_name:
                item_id = i_id
                break
        if item_id is not None:
            # Remove from location
            game_state[location].remove(item_id)
            # Put in inventory
            user['inventory'][item_id] = 'inventory'

            # Notify hub about the ownership transfer
            if hub_url:
                transfer_data = {
                    "domain": domain_id,
                    "secret": secret,
                    "user": user_id,
                    "item": item_id,
                    "to": "inventory"
                }
                try:
                    async with app.client.post(f"{hub_url}/transfer", json=transfer_data) as resp:
                        if resp.status != 200:
                            error_resp = await resp.text()
                            print(f"Failed to transfer item to hub: {resp.status}, {error_resp}")
                except Exception as e:
                    print(f"Exception while transferring item to hub: {e}")

            return f"You have taken the {item_map[item_id]['name']}."
        else:
            return f"There is no {item_name} here to take."

    elif verb == "inventory":
        # Check items in user's inventory
        carried = [item_map[i_id]['name'] for i_id, loc in user['inventory'].items() if loc == 'inventory']
        if carried:
            return f"You are carrying: {', '.join(carried)}."
        else:
            return "You are not carrying anything."

    elif verb == "drop":
        if not args:
            return "Drop what?"
        item_name = ' '.join(args).lower()
        item_id = None
        # Find the item in the user's inventory
        for i_id, loc in user['inventory'].items():
            if loc == 'inventory' and item_map[i_id]['name'].lower() == item_name:
                item_id = i_id
                break
        if item_id is not None:
            # Remove from inventory
            user['inventory'][item_id] = location
            # Add to current location
            if location not in game_state:
                game_state[location] = []
            game_state[location].append(item_id)

            # Notify hub about the ownership transfer
            if hub_url:
                transfer_data = {
                    "domain": domain_id,
                    "secret": secret,
                    "user": user_id,
                    "item": item_id,
                    "to": location  # Dropping to a location
                }
                try:
                    async with app.client.post(f"{hub_url}/transfer", json=transfer_data) as resp:
                        if resp.status != 200:
                            error_resp = await resp.text()
                            print(f"Failed to transfer item to hub: {resp.status}, {error_resp}")
                except Exception as e:
                    print(f"Exception while transferring item to hub: {e}")

            return f"You have dropped the {item_map[item_id]['name']}."
        else:
            return f"You don't have a {item_name} to drop."

    elif verb == "use":
        if not args:
            return "Use what?"
        if len(args) == 1:
            return "Use what on what?"
        item_name = args[0].lower()
        target_name = ' '.join(args[1:]).lower()
        item_id = None
        target_id = None

        # Find the item to use
        for i_id, loc in user['inventory'].items():
            if loc == 'inventory' and item_map[i_id]['name'].lower() == item_name:
                item_id = i_id
                break
        if not item_id:
            return f"You don't have a {item_name} to use."

        # Find the target item in the current location
        for i_id in game_state.get(location, []):
            if item_map[i_id]['name'].lower() == target_name:
                target_id = i_id
                break
        # If not found, check in inventory
        if not target_id:
            for i_id, loc in user['inventory'].items():
                if loc == 'inventory' and item_map[i_id]['name'].lower() == target_name:
                    target_id = i_id
                    break
        if not target_id:
            return f"There is no {target_name} here to use the {item_name} on."

        # Define specific interactions
        if item_map[item_id]['name'].lower() == "key" and item_map[target_id]['name'].lower() == "cabinet":
            cabinet_state = game_state.get('cabinet_state', 'locked')
            if cabinet_state == 'locked':
                # Check if user has the key
                if item_id in user['inventory'] and user['inventory'][item_id] == 'inventory':
                    game_state['cabinet_state'] = 'closed'
                    # Optionally, remove key from inventory if it's single-use
                    # del user['inventory'][item_id]
                    return "You use the key to unlock the cabinet doors."
                else:
                    return "You don't have a key to use."
            elif cabinet_state == 'closed':
                game_state['cabinet_state'] = 'locked'
                return "You use the key to lock the cabinet doors."
            elif cabinet_state == 'open':
                return "The cabinet is already open."

        elif item_map[item_id]['name'].lower() == "switch":
            # User uses the switch
            switch_state = game_state.get('switch_state', 'down')
            if switch_state == 'down':
                game_state['switch_state'] = 'up'
                # Change screen state to 'password'
                game_state['screen_state'] = 'password'
                return "You toggle the switch to the up position."
            else:
                game_state['switch_state'] = 'down'
                # Change screen state to 'blank'
                game_state['screen_state'] = 'blank'
                return "You toggle the switch to the down position."

        elif item_map[item_id]['name'].lower() == "paper" and item_map[target_id]['name'].lower() == "sign":
            return "You unfold the paper and compare it to the sign."

        elif item_map[item_id]['name'].lower() == "axe" and item_map[target_id]['name'].lower() == "cabinet":
            return "You try to use the axe on the cabinet, but it's too sturdy."

        elif item_map[item_id]['name'].lower() == "i-card" and item_map[target_id]['name'].lower() == "terminal":
            return "You swipe the i-card at the terminal, granting you access."

        else:
            return f"You can't use the {item_map[item_id]['name']} on the {item_map[target_id]['name']}."

    elif verb == "open":
        if not args:
            return "Open what?"
        item_name = ' '.join(args).lower()
        if item_name != 'cabinet':
            return f"You can't open the {item_name}."
        cabinet_state = game_state.get('cabinet_state', 'locked')
        if cabinet_state == 'locked':
            return "The cabinet is locked."
        elif cabinet_state == 'closed':
            # Open the cabinet
            game_state['cabinet_state'] = 'open'
            # Add depth-1 items to podium
            depth1_items = [item_id for item_id, item in item_map.items() if item.get('depth') == 1 and item_id not in game_state['podium']]
            game_state['podium'].extend(depth1_items)
            return "You open the cabinet. Inside you see a switch."
        elif cabinet_state == 'open':
            return "The cabinet is already open."

    elif verb == "close":
        if not args:
            return "Close what?"
        item_name = ' '.join(args).lower()
        if item_name != 'cabinet':
            return f"You can't close the {item_name}."
        cabinet_state = game_state.get('cabinet_state', 'locked')
        if cabinet_state == 'locked':
            return "The cabinet is already locked."
        elif cabinet_state == 'closed':
            return "The cabinet is already closed."
        elif cabinet_state == 'open':
            # Close the cabinet
            game_state['cabinet_state'] = 'closed'
            # Remove depth-1 items from podium
            depth1_items = [item_id for item_id, item in item_map.items() if item.get('depth') == 1]
            for item_id in depth1_items:
                if item_id in game_state['podium']:
                    game_state['podium'].remove(item_id)
            return "You close the cabinet."

    elif verb == "look":
        if not args:
            # Return exact description based on location
            description = location_descriptions.get(location, "You are nowhere.")
            # If location is 'podium' and cabinet is open, include switch and screen
            if location == 'podium':
                cabinet_state = game_state.get('cabinet_state', 'locked')
                if cabinet_state == 'open':
                    # Ensure switch is present
                    switch_id = next((id for id, item in item_map.items() if item['name'].lower() == 'switch'), None)
                    if switch_id and switch_id not in game_state['podium']:
                        game_state['podium'].append(switch_id)
                    # Ensure screen is present
                    screen_id = next((id for id, item in item_map.items() if item['name'].lower() == 'screen'), None)
                    if screen_id and screen_id not in game_state['podium']:
                        game_state['podium'].append(screen_id)
                else:
                    # Cabinet is not open; ensure screen is present but its state is handled separately
                    screen_id = next((id for id, item in item_map.items() if item['name'].lower() == 'screen'), None)
                    if screen_id and screen_id not in game_state['podium']:
                        game_state['podium'].append(screen_id)

            # Exclude hidden items from the visible items list
            items_here = [item_map[i_id]['name'] for i_id in game_state.get(location, []) if not item_map[i_id].get('hidden', False)]
            if items_here:
                description += f" You see the following items: {', '.join(items_here)}."
            else:
                description += " There are no visible items here."
            return description
        else:
            item_name = ' '.join(args).lower()
            if item_name == 'cabinet':
                cabinet_state = game_state.get('cabinet_state', 'locked')
                if cabinet_state == 'locked':
                    return "The cabinet is locked."
                elif cabinet_state == 'closed':
                    return "The cabinet is closed."
                elif cabinet_state == 'open':
                    return "The cabinet is open."
                else:
                    return "The cabinet is in an unknown state."
            elif item_name == 'switch':
                if location == 'podium' and game_state.get('cabinet_state') == 'open':
                    switch_state = game_state.get('switch_state', 'down')
                    return f"The switch has a small label reading 'power' and is in the {switch_state} position."
                else:
                    return "There is no switch here."
            elif item_name == 'screen':
                if location == 'podium':
                    screen_state = game_state.get('screen_state', 'blank')
                    if screen_state == 'blank':
                        return "The screen is blank."
                    elif screen_state == 'password':
                        return "The screen displays a password prompt."
                    elif screen_state == 'won':
                        return "The screen displays: Congratulations! You've won!"
                    else:
                        return "The screen is in an unknown state."
                else:
                    return "There is no screen here."
            else:
                # Look for the item
                item_id = None
                # Check in user's inventory
                for i_id, loc in user['inventory'].items():
                    if item_map[i_id]['name'].lower() == item_name:
                        item_id = i_id
                        break
                # If not found, check in current location
                if not item_id:
                    for i_id in game_state.get(location, []):
                        if item_map[i_id]['name'].lower() == item_name:
                            item_id = i_id
                            break
                if item_id is not None:
                    # If the item is 'screen', adjust description based on state
                    if item_map[item_id]['name'].lower() == 'screen':
                        screen_state = game_state.get('screen_state', 'blank')
                        if screen_state == 'blank':
                            return "The screen is blank."
                        elif screen_state == 'password':
                            return "The screen displays a password prompt."
                        elif screen_state == 'won':
                            return "The screen displays: Congratulations! You've won!"
                        else:
                            return "The screen is in an unknown state."
                    return item_map[item_id]['description']
                else:
                    return f"There is no {item_name} here."

    elif verb == "read":
        if not args:
            return "Read what?"
        item_name = ' '.join(args).lower()
        item_id = None
        # Check in user's inventory
        for i_id, loc in user['inventory'].items():
            if item_map[i_id]['name'].lower() == item_name:
                item_id = i_id
                break
        # If not found, check in current location
        if not item_id:
            for i_id in game_state.get(location, []):
                if item_map[i_id]['name'].lower() == item_name:
                    item_id = i_id
                    break
        if item_id and 'read' in item_map[item_id].get('verb', {}):
            return item_map[item_id]['verb']['read']
        else:
            return f"You can't read the {item_name}."

    elif verb == "go":
        if not args:
            return "Go where?"
        direction = args[0].lower()
        if direction == 'east':
            return "$journey east"

        directions = {
            "north": "classroom",
            "south": "foyer",
            "west": "podium",
            "up": "classroom",
            "down": "podium",
        }

        new_location = directions.get(direction)
        if not new_location:
            return f"You can't go {direction} from here."
        user['location'] = new_location

        description = location_descriptions.get(new_location, "You are nowhere.")

            if switch_id and switch_id not in game_state['podium']:

            if items_here:
                description += f" You see the following items: {', '.join(items_here)}."
        else:
            # Exclude hidden items from the visible items list
            items_here = [item_map[i_id]['name'] for i_id in game_state.get(new_location, []) if not item_map[i_id].get('hidden', False)]
            if items_here:
                description += f" You see the following items: {', '.join(items_here)}."
            else:

    elif verb == "tell":

        else:
            return f"You can't tell the {target}."

    else:
        return f"Unknown command: {verb}"

async def arrive(uid: int, dest: int, app:web.Application) -> None:
    """Alert a domain that a user has arrived"""
    owned, carried, dropped, prize = [],[],[],[]
    for tid, loc in users[uid]['inventory'].items():
        t = templates[tid]
        brief = {k:v for k,v in t.items() if k in ('name','description','verb')}
        brief['id'] = tid
        if loc == 'inventory':
            if t['home'] == dest:
                owned.append(brief)
            else: carried.append(brief)
        elif loc[0] == dest:
            brief['location'] = loc[1]
            dropped.append(brief)
    for tid in domains[dest]['loot']:
        if tid not in users[uid]['inventory']:
            t = templates[tid]
            brief = {k:v for k,v in t.items() if k in ('name','description','verb','depth')}
            brief['id'] = tid
            prize.append(brief)
    
    try:
        async with app.client.post(domains[dest]['url']+'/arrive', json={
            'secret':domains[dest]['secret'],
            'user':uid,
            'owned':owned,
            'carried':carried,
            'dropped':dropped,
            'prize':prize,
        }) as resp:
            assert resp.status == 200, (resp.status, await resp.read())
    except BaseException as ex:
        print('ERROR:',domains[dest]['url']+'/arrive','did not work',repr(ex))

@routes.post("/transfer")
async def transfer(req: web.Request) -> web.Response:
    """Called by domain servers to move items into our out of gear
    
    { "domain": sending domain's id
    , "secret": sending domain's secret id
    , "user": user id
    , "item": item type id
    , "to": destination
    }
    
    Destination "inventory" means the item should be carried.
    Any other destination names some location within the sending domain (as if dropped).
    
    """
    try: data = await req.json()
    except: return web.json_response(status=400, data={"error":"JSON data required"})
    did = checkdid(data)
    if isinstance(did, web.Response): return did
    uid = data.get('user')
    if uid not in users:
        return web.json_response(status=400, data={"error":"Valid user ID required"})
    tid = data.get('item')
    if tid not in templates:
        return web.json_response(status=400, data={"error":"Valid item ID required"})
    if 'to' not in data:
        return web.json_response(status=400, data={"error":"Missing \"to\" field"})
    
    old = users[uid]['inventory'].get(tid)
    new = data['to']
    owned = templates[tid]['home'] == did or did in templates[tid].get('hosts',[])
    
    
    if old == new:
        return web.json_response(status=409, data={"error":"Cannot move item to where it already is"})
    
    if old is None and not owned:
        return web.json_response(status=403, data={"error":"Cannot generate items that don't belong to you"})
    if old is not None and new != 'inventory' and templates[tid]['home'] != did:
        return web.json_response(status=403, data={"error":"Cannot move or remove items that don't belong to you"})

    if old is not None and old[0] != did:
        return web.json_response(status=403, data={"error":"That item has been dropped in a different domain"})

    users[uid]['inventory'][tid] = new if new == 'inventory' else (did, new)


    return web.json_response(status=200, data={"ok":"Item transferred"})


@routes.post("/query")
async def transfer(req: web.Request) -> web.Response:
    """Called by domain servers to find out what items are where
    
    { "domain": sending domain's id
    , "secret": sending domain's secret id
    , "user": user id
    , "location": either "inventory" or a domain-selected location value
    , "depth": asks for any prizes of that depth
    }
    
    One (not both) of "location" and "depth" must be provided.
    
    Return is a list of item ID.
    """

    except: return web.json_response(status=400, data={"error":"JSON data required"})

        return web.json_response(status=400, data={"error":"Must provide location xor depth"})

    if 'location' in data:

        if where is None:
            return web.json_response(status=400, data={"error":"Location required"})
        if where != 'inventory':
            where = (did, where)

    else:
        resp = [iid for iid in domains[did]['loot'] if iid not in users[uid]['inventory'] and templates[iid].get('depth') == data['depth']]
