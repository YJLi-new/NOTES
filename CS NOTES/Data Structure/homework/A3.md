**Possible Solutions**

Breadth-First Search (BFS): Using BFS to calculate the shortest path from each node to every other node. This method is straightforward but could be less efficient for larger trees due to repetitive calculations.

Centroid Decomposition: Decompose the tree into a centroid tree and calculate distances using properties of the centroid, potentially reducing the complexity of distance calculations by focusing on smaller subtrees.

**Solution Description**

The implemented solution leverages Tree Dynamic Programming (Tree DP) through two depth-first search (DFS) passes:

DFS1: This phase gathers subtree-specific data, including the count of black nodes within each subtree (subtree_black_num) and a weighted count that aggregates the distances within the subtree (subtree_num).

DFS2: In the second phase, the function adjusts the values computed in DFS1 to include contributions from the entire tree by considering nodes outside the current subtree. It utilizes data passed down from the parent to update the node-specific results.

This method employs a recursive approach, where each node processes its children before integrating the results into its own computation. The primary computations involve:

Accumulating subtree statistics during the first DFS to factor in local contributions from children nodes.

Adjusting global contributions during the second DFS to integrate external contributions effectively.

The outputs for each node are calculated modulo 10^9+7 to maintain manageable number sizes.

**Why This Solution is Superior**

Optimal Complexity and Efficiency: Unlike BFS, which might require recalculating distances for each node exhaustively, Tree DP efficiently partitions calculations into manageable components using subtree properties. Tree DP ensures that each node and edge is processed a minimal number of times (specifically, twice due to the two DFS passes), achieving an optimal time complexity of O(n).

Modularity and Clarity: The separation into two distinct DFS passes enhances the modularity of the code. The first pass is dedicated to local data gathering, and the second focuses on integrating this data with global context, which clarifies the flow and purpose of computations.

```cpp
#include <iostream>
#include <string>
#include <vector>

using namespace std;

vector<int> scoreCounting(string s) {

	vector<int> count = { 0,0 };

	if (s == "O") {
		return vector<int> { -1,-1 };
	}

	for (char c : s) {
		if (c == 'A') {
			count[1]++;
		}
		else if (c == 'O') {
			count[0]++;
		}
	}
	return count;
}

int main() {
	string result_s;

	getline(cin, result_s);

	vector<int> count = scoreCounting(result_s);
	int score_A = count[1];

	if (score_A == -1) {
		cout << "A:0" << endl;
		cout << "B:0" << endl;
		cout << "Draw!" << endl;
	}

	else {
		int length_s = result_s.length() - count[0];

		if (length_s - score_A == score_A) {
			cout << "A:" << score_A << endl;
			cout << "B:" << score_A << endl;
			cout << "Draw!" << endl;
}

		else if (2 * score_A > length_s) {
			cout << "A:" << score_A << endl;
			cout << "B:" << length_s - score_A << endl;
			cout << "A wins!" << endl;
}

		else {
			cout << "A:" << score_A << endl;
			cout << "B:" << length_s - score_A << endl;
			cout << "B wins!" << endl;
	}
	}



	return 0;
}
```

```cpp
#include <iostream>
#include <vector>
#include <utility>
#include <unordered_map>

using namespace std;

void DFS1(int curr, int parent,
    vector<unordered_map<int, int>>& adj_list, vector<int>& color_list, vector<long long>& subtree_num, vector<long long>& subtree_black_num, vector<long long>& total_black_weighs, int n) {

    subtree_black_num[curr] += color_list[curr];
    subtree_num[curr] = 0;

    for (auto edge : adj_list[curr]) {
        int next = edge.first;
        int weight = edge.second;

        if (next != parent) {
            DFS1(next, curr, 
                adj_list, color_list, subtree_num, subtree_black_num, total_black_weighs, n);

            subtree_black_num[curr] += subtree_black_num[next];
            subtree_num[curr] += subtree_num[next] + subtree_black_num[next] * weight;
        }
    }
}

void DFS2(int curr, int parent, 
    vector<unordered_map<int, int>>& adj_list, vector<int>& color_list, vector<long long>& sub_tree_num, vector<long long>& sub_tree_black_num, vector<long long>& total_black_weighs, int total_black_num) {
    if (parent != -1) {
        int weight = adj_list[parent][curr];
        long long out_black_num = total_black_num - sub_tree_black_num[curr];
        long long out_subtree_wei = total_black_weighs[parent] - (sub_tree_num[curr] + sub_tree_black_num[curr] * weight);

        total_black_weighs[curr] = sub_tree_num[curr] + out_subtree_wei + out_black_num * weight;
    }
    else {
        total_black_weighs[curr] = sub_tree_num[curr];
    }

    for (auto edge : adj_list[curr]) {
        int next = edge.first;

        if (next != parent) {
            DFS2(next, curr, 
                adj_list, color_list, sub_tree_num, sub_tree_black_num, total_black_weighs, total_black_num);
        }
    }
}

int main() {
    int n;
    cin >> n;

    vector<unordered_map<int, int>> adj_list(n + 1);
    vector<int> color_list(n + 1, 0);
    vector<long long> sub_tree_num(n + 1, 0);
    vector<long long> sub_tree_black_num(n + 1, 0);
    vector<long long> total_black_weighs(n + 1, 0);

    for (int i = 1; i <= n; i++) {
        cin >> color_list[i];
    }

    for (int i = 1; i < n; i++) {
        int u, v, w;
        cin >> u >> v >> w;
        adj_list[u][v] = w;
        adj_list[v][u] = w;
    }

    DFS1(1, -1, adj_list, color_list, sub_tree_num, sub_tree_black_num, total_black_weighs, n);
    DFS2(1, -1, adj_list, color_list, sub_tree_num, sub_tree_black_num, total_black_weighs, sub_tree_black_num[1]);

    for (int i = 1; i <= n; i++) {
        cout << total_black_weighs[i] % (1000000007) << endl;
    }

    return 0;
}
```
