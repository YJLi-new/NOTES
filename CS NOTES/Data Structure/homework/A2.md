Problem 3

**Possible Solutions**

1\. A recursive approach that tries removing each digit in turn and checking the resulting number, recursively continuing selecting the smallest result.

2\. Using a dynamic approach to memorize intermediate results for subsets of the sequence and the number of digits removed.

**Solution Description**

The code reads a sequence of digits as a string and an integer m, representing the number of digits to remove to form the smallest possible number. It iterates through the given sequence, maintaining a result list. For each digit, if the current digit is less than the last digit of the result list and there are still digits to remove (rest_nums \> 0), it removes the last digit of the result list. This ensures the resulting number is as small as possible by prioritizing lower digits earlier in the sequence. The final result is printed, omitting leading zeros unless the result is solely zeros.

**Why This Solution is Superior**

1\. Efficiency: The algorithm efficiently reduces the sequence to the required length in a single pass, with a time complexity of O(n), where n is the length of the input sequence.

2\. Readability & Maintainability: The code structure is straightforward, making it easy to understand and maintain. The logic is implemented in a single loop with clear conditions for removing digits, enhancing code clarity.

#include <iostream>
#include <vector>
#include <string>
#include <sstream>

using namespace std;

int findWinner(int n, vector<vector<int>> M) {
	int result = -1;

	for (int i = 0; i < n; i++) {
		bool is_candidate = true;
		for (int j = 0; j < n; j++) {
			if (M[i][j] == 1) {
				is_candidate = false;
			}
		}
		if (is_candidate) {
			result = i;
			break;
		}
	}

	if (result == -1) {
		return -1;
	}

	for (int i = 0; i < n && i != result; i++) {
		if (M[i][result] == 0) {
			return -1;
		}
	}

	return result;
}

int main() {
	string line;
	vector<vector<int>> M;

	getline(cin, line);
	int n = stoi(line);

	while (getline(cin, line)) {
		vector<int> pp_ith(n,0);
		istringstream iss(line);

		for (int j = 0; j < n; j++) {
			iss >> pp_ith[j];
		}
		M.push_back(pp_ith);
	}
	cout << findWinner(n, M) << endl;

	return 0;
}

void swapListRow(vector<vector<int>> &needed_list, int t1, int t2) {
	vector<int> temp1 = needed_list[t1];
	vector<int> temp2 = needed_list[t2];
	needed_list[t1] = temp2;
	needed_list[t2] = temp1;
}

int partitionQS(vector<vector<int>> &needed_list, int l, int r, int p) {
	swapListRow(needed_list, r, p);

	int next = l;

	for (int i = l; i < r; i++) {
		if (needed_list[i][0] > needed_list[r][0]) {
			swapListRow(needed_list, next, i);
			next += 1;
		}
	}
	swapListRow(needed_list, next, r);
	return next;
}

int quickSort(vector<vector<int>> &needed_list, int l, int r) {
	if (l >= r) {
		return 0;
	}
	int p = (l + r) / 2;
	int new_p = partitionQS(needed_list, l, r, p);
	quickSort(needed_list, l, new_p - 1);
	quickSort(needed_list, new_p + 1, r);

	return 0;
}

bool compareList(vector<int>& l1, vector<int>& l2) {

}

vector<int> findHighestList(int n, int k, vector<int> house_list) {

		sorted_houses.push_back(vector<int> { house_list[i], i });

		int curr_house_height = 0;

using namespace std;

vector<int> encryptScroll(int n, vector<int> seq, int m) {
    vector<int> result;
    int rest_nums = n - m;

    for (int num : seq) {
        while (rest_nums > 0 && !result.empty() && num < result.back()) {
            result.pop_back();
            rest_nums -= 1;
        }
        result.push_back(num);
    }

    return result;
}

int main() {

	int m;
