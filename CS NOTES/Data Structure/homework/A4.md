#include <iostream>
#include <vector>
#include <string>
#include <algorithm>

using namespace std;

struct Node {
	int value;
	vector<Node*> children_set;
	Node(int value) : value(value), children_set{} {};
	void addChild(Node* n) {
		children_set.push_back(n);
	}
};

bool DFSVisit(int& count, vector<vector<int>>& tree, int node, vector<int>& visited, vector<int>& N_permutation) {
	if (N_permutation[count] != node) {
		return false;
	}
	visited[node] = 1;
	count++;

	for (int child: tree[node]) {
		if (visited[child] == 0) {
			if (!DFSVisit(count, tree, child, visited, N_permutation)) {
				return false;
			}
		}
	}

	return true;

}

bool DFS(int N, vector<vector<int>>& tree, vector<int>& N_permutation) {
	bool result = true;
	vector<int> visited(N + 1, 0);
	int count = 0;

	result = DFSVisit(count, tree, 1, visited, N_permutation);

	return result;
}

int main() {
	int T;
	cin >> T;

	int N = 0;

	for (int i = 0; i < T; i++) {
		cin >> N;
		vector<vector<int>> tree(N + 1);

		for (int child = 2; child <= N; child++){
			int parent;
			cin >> parent;
			tree[parent].push_back(child);
		}

		vector<int> N_permutation{};
		vector<int> map_permutation(N + 1);
		for (int j = 1; j <= N; j++) {
			int N_permutation_int;
			cin >> N_permutation_int;
			N_permutation.push_back(N_permutation_int);
			map_permutation[N_permutation_int] = j;
		}

		for (vector<int>& parent : tree) {
			sort(parent.begin(), parent.end(),
				[&](int a_child, int b_child) 
				{ return map_permutation[a_child] < map_permutation[b_child]; });
		}

		cout << (DFS(N, tree, N_permutation) ? "YES" : "NO") << endl;
	}

	return 0;
}

#include <iostream>
#include <vector>
#include <queue>
#include <climits>

using namespace std;

struct CompareTwoPairs {
    bool operator()(const pair<int, int>& a, const pair<int, int>& b) {
        return a.first > b.first;
    }
};

int primMST(int N, int W, vector<vector<int>>& A) {
    priority_queue<pair<int, int>, vector<pair<int, int>>, CompareTwoPairs> pq;
    vector<int> add_price(N, INT_MAX);
    vector<int> parent(N, -1);
    vector<bool> is_in(N, false);

    int first_gift = 0;
    add_price[first_gift] = 0;
    pq.push({ 0, first_gift });

    int cost = W;

    while (!pq.empty()) {
        int u = pq.top().second;
        pq.pop();

        if (is_in[u]) { continue; }

        is_in[u] = true;

        for (int v = 0; v < N; v++) {
            if (!is_in[v] && A[u][v] != 0 && A[u][v] < add_price[v]) {
                parent[v] = u;
                add_price[v] = A[u][v];
                pq.push({ add_price[v], v });
            }
        }
    }

    for (int v = 1; v < N; v++) {
        if (parent[v] != -1 && add_price[v]<W) {
            cost += add_price[v];
        }
        else {
            cost += W;
        }
    }

    return cost;
}

int main() {
	int N, W;
	cin >> N;
	cin >> W;

	if (W == 0 || N == 0) {
		cout << 0 << endl;
        return 0;
	}

	vector<vector<int>> A(N, vector<int>(N));

	for (int i = 0; i < N; i++) {
		for (int j = 0; j < N; j++) {
			int a;
			cin >> a;
			A[i][j] = a;
		}
	}

    cout << primMST(N, W, A) << endl;

	return 0;
}
