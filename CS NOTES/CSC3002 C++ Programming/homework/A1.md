/* Function prototypes */

int fact(int n);

/*
 * Implementation notes: permutations
 * ----------------------------------
 * This function takes advantage of the property that n! divided by (n - k)!
 * is simply the first k terms in the product
 *
 *     n * (n - 1) * (n - 2) * . . .
 */

int permutations(int n, int k) {
    int result = 1;

    if (k>n || n<0 || k<0) {
        cout << "Please enter n >= k >= 0" << endl;
        result = 0;
    }

    // P = n·...·(n-r+1)
    else {
        for (int i=n-k+1; i<n+1; i++) {
            result = result * i;
        }
    }

    return result;
}

/*
 * Implementation notes: combinations
 * ----------------------------------
 * This function returns the permutations value divided by k!, which
 * is computed by the function fact.
 */

int combinations(int n, int k) {
    int result = 1;

    // If r>n, then C=0
    if (k > n) {
        cout << "Please enter n >= k >= 0" << endl;
        result = 0;
    }

    // If r<=n, then calculate C normally
    else {
        int result1 = fact(k);
        int result2 = 1;

        for (int i = k + 1; i < n + 1; i++) {
            result2 = result2 * i;
        }

        result = result2 / result1;
    }

    return result;
}

/*
 * Function: fact(n)
 * Usage: int result = fact(n);
 * ----------------------------
 * Returns the factorial of n, which is the product of all the
 * integers between 1 and n, inclusive.
 */

int fact(int n) {
    int result = 1;

    if (n < 0) {
        cout << "A negative number does not have a factorial." << endl;
        result = 0;
    }

    else if (n > 1) {
        for (int i=1; i<n+1; i++) {
            result = result * i;
        }
    }

    // else if (n == 1 || n == 0) {result = 1;}

    return result;
}

 * Function: findDNAMatch
 * Usage: int pos = findDNAMatch(s1, s2);
 *        int pos = findDNAMatch(s1, s2, start);
 * ---------------------------------------------
 * Returns the first index position at which strand s1 would bind to
 * the strand s2, or -1 if no such position exists.  If the start
 * parameter is supplied, the search begins at that index position.
 */

int findDNAMatch(string s1, string s2, int start) {
    int found_pos = start;
    int leng1 = s1.length();

    for (int i=0; i<leng1; i++) {
        if (matchingStrand(s1)[i] != s2[start+i] || matchingStrand(s1)[i] == 'X') {
            found_pos = -1;
            break;
        }
    }

    return found_pos;
}

/*
 * Function: matchingStrand
 * Usage: string match = matchingStrand(strand);
 * ---------------------------------------------
 * Returns a string in which each base has been replaced by its
 * counterpart base (C <-> G and A <-> T).  Any other characters
 * are replaced by an X.
 */

string matchingStrand(string strand) {
    int leng = strand.length();
    string counter_strand = "";
    map<char, char> geneMaps = {
        {'A', 'T'},
        {'C', 'G'},
        {'T', 'A'},
        {'G', 'C'},
        {'X', 'X'}
    };

    string exist_gene = "ACTG";

    for (int i=0; i < leng; i++) {

        auto counter_gene = geneMaps.find(strand[i]);
        char cg;
        if (counter_gene == geneMaps.end()) {
            cg = 'X';
        }
        else {
            cg = counter_gene->second;
        }


        counter_strand += cg;
    }

    return counter_strand;
}

/*
 * Function: findAllMatches
 * Usage: findAllMatches(s1, s2);
 * ------------------------------
 * Finds all positions at which s1 can bind to s2.
 */

    int available_length = s2.length() - s1.length();

    if (available_length >= 0) {
        bool has_pos = false;

                cout << pos << endl;
            }
        }
        if (!has_pos) {
            cout << s1 << " has no matches in " << s2;
        }
    }
}

/*
 * File: RemoveComments.cpp
 * ------------------------
 * Prints out a file after removing comments.
 */
// header file for local test
#include <iostream>
#include <iomanip>
//#include "CSC3002OJActive/assignment1/RemoveComments.h" //for OJ
//#include "RemoveComments.h" //for local test
using namespace std;

#include <string>
#include <queue>

void removeComments(istream& is, ostream& os) {
    string line;
    bool in_comment = false;
    string result;

    while (getline(is, line)) {
        if (in_comment) {
            size_t end_comment_pos = line.find("*/");
            if (end_comment_pos != string::npos) {
                line = line.substr(end_comment_pos + 2);
                in_comment = false;
                result += "\n";
            } else {
                continue;
            }
        }

        size_t start_comment_pos = line.find("//");
        if (start_comment_pos != string::npos) {
            line = line.substr(0, start_comment_pos);
            result += "\n";
            continue;
        }

        start_comment_pos = line.find("/*");
        while (start_comment_pos != string::npos) {
            size_t end_comment_pos = line.find("*/", start_comment_pos);
            if (end_comment_pos != string::npos) {
                line = line.substr(0, start_comment_pos) + line.substr(end_comment_pos + 2);
                start_comment_pos = line.find("/*", start_comment_pos);
            } else {
                line = line.substr(0, start_comment_pos);
                in_comment = true;
                break;
            }
        }

        result += line;
        if (!line.empty() && !in_comment) {
            result += "\n";
        }
    }

    os << result;
}


void TestRemoveComments(){
    removeComments(cin, cout);
}

// DO NOT modify the main() function
int main(int argc, char* argv[]) {
    TestRemoveComments();
    return 0;
}
