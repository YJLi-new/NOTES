StringMap::StringMap() {
   nBuckets = INITIAL_BUCKET_COUNT;
   buckets = new KeyValuePair[nBuckets];
   clear();
}

StringMap::~StringMap() {
   delete[] buckets;

/*
 * Implementation notes: get
 * -------------------------
 * The get method calls findKeyValuePair to search the linked list for the
 * matching key.  If no key is found, get returns the empty string.
 */

string StringMap::get(const string & key) const {
   int index = findKey(key);
   return (index == -1) ? "" : buckets[index].value;
}

/*
 * Implementation notes: put
 * -------------------------
 * The put method calls insertKey to search the bucket array for a matching
 * key.  If a key already exists, put simply resets the value field.  If no
 * matching key is found, put adds a new entry in the first free slot.
 */

void StringMap::put(const string & key, const string & value) {
   if ((double) count / nBuckets > REHASH_THRESHOLD) {
      rehash(2 * nBuckets + 1);
   }
   int index = insertKey(key);
   buckets[index].value = value;
}

int StringMap::size() const {
   return count;
}

bool StringMap::isEmpty() const {
   return count == 0;
}

bool StringMap::containsKey(const string & key) const {
   return findKey(key) != -1;
}

/*
 * Implementation notes: remove
 * ----------------------------
 * This implementation is tricky because removing one key can make later
 * keys inaccessible.  This implementation finds the first key that could
 * have gone in this position (if any) and moves it to this space, repeating
 * that process until an empty entry is found.  A much simpler but less
 * efficient strategy is to rehash after every deletion.
 */

void StringMap::remove(const string & key) {
   int index = findKey(key);
   if (index != -1) {
      buckets[index].occupied = false;
      count--;
      int toFill = index;
      while (true) {
         index = (index + 1) % nBuckets;
         if (!buckets[index].occupied) return;
         if (insertKey(buckets[index].key) == toFill) {
            buckets[toFill].value = buckets[index].value;
            buckets[index].occupied = false;
            toFill = index;
         }
      }
   }
}

void StringMap::clear() {
   for (int i = 0; i < nBuckets; i++) {
      buckets[i].occupied = false;
   }
   count = 0;
}

int StringMap::getNBuckets() {
   return nBuckets;
}

/* TODO PART:
 * Implementation notes: rehash
 * ----------------------------
 * The rehash method iterates over the existing key-value pairs, entering
 * them into a new table.
 */

void StringMap::rehash(int nBuckets) {
    KeyValuePair* old_buckets;
    old_buckets = new KeyValuePair[this->nBuckets];
    for (int i = 0; i < this->nBuckets; i++) {
        old_buckets[i] = buckets[i];
    }

    delete[] buckets;
    int ori_nBuckets = this->nBuckets;
    this->nBuckets = nBuckets;

    buckets = new KeyValuePair[nBuckets];
    clear();

    for (int i = 0; i < ori_nBuckets; i++) {
        if (old_buckets[i].occupied) {
            int index = insertKey(old_buckets[i].key);
            buckets[index].value = old_buckets[i].value;
            buckets[index].occupied = true;
        }
    }

    delete[] old_buckets;
}

/*
 * Private method: findKey
 * Usage: int index = findKey(key);
 * --------------------------------
 * This private method looks for a key in the buckets array.  If the
 * key is found, findKey returns its index.  If no match is found, the
 * findKey returns -1.
 */

int StringMap::findKey(const string & key) const {
    
    int hc_index = (hashCode(key) % nBuckets);
    while (true) {
        if (buckets[hc_index].occupied) {
            if (buckets[hc_index].key == key) {
                return hc_index;
            }
        }
        hc_index = (1 + hc_index) % nBuckets;
        if (hc_index == (hashCode(key) % nBuckets)) { return -1; }
    }
}

/*
 * Private method: insertKey
 * Usage: int index = insertKey(key);
 * ----------------------------------
 * This private method is identical to findKey except that it inserts the
 * key in the correct place if it is not already in the table.
 */

int StringMap::insertKey(const string & key) {
    int hc_index = findKey(key);

    if (hc_index == -1) {
        hc_index = (hashCode(key) % nBuckets);
        while (buckets[hc_index].occupied) {

            hc_index = (1 + hc_index) % nBuckets;
            if (hc_index == (hashCode(key) % nBuckets)) { return -1; }
        }

        buckets[hc_index].key = key;
        buckets[hc_index].occupied = true;
        count++;
        }

    return hc_index;
}

/*
 * Implementation notes: hashCode
 * ------------------------------
 * This function takes a string key and uses it to derive a hash code,
 * which is nonnegative integer related to the key by a deterministic
 * function that distributes keys well across the space of integers.
 * The specific algorithm used here is called djb2 after the initials
 * of its inventor, Daniel J. Bernstein, Professor of Mathematics at
 * the University of Illinois at Chicago.
 */

const int HASH_SEED = 5381;               /* Starting point for first cycle */
const int HASH_MULTIPLIER = 33;           /* Multiplier for each cycle      */
const int HASH_MASK = unsigned(-1) >> 1;  /* The largest positive integer   */

int hashCode(const string & str) {
   unsigned hash = HASH_SEED;
   int n = str.length();
   for (int i = 0; i < n; i++) {
      hash = HASH_MULTIPLIER * hash + str[i];
   }
   return int(hash & HASH_MASK);
}

template <typename ValueType>
void PriorityQueue<ValueType>::enqueue(ValueType value, double priority) {
    if (heap.empty()) {
        count = 0;
        enqueueCount = 0;
    }

    enqueueHeap(value, priority);
}

/*
 * Implementation notes: dequeue, peek, peekPriority
 * -------------------------------------------------
 * These methods must check for an empty queue and report an error if there
 * is no first element.
 */

template <typename ValueType>
ValueType PriorityQueue<ValueType>::dequeue() {
   if (count == 0) error("dequeue: Attempting to dequeue an empty queue");

   ValueType value = dequeueHeap();

   return value;
}

template <typename ValueType>
ValueType PriorityQueue<ValueType>::peek() const {
   if (count == 0) error("peek: Attempting to peek at an empty queue");

   return heap[0].value;
}

template <typename ValueType>
double PriorityQueue<ValueType>::peekPriority() const {
   if (count == 0) error("peekPriority: Attempting to peek at an empty queue");

   return heap[0].priority;
}

template <typename ValueType>
void PriorityQueue<ValueType>::enqueueHeap(ValueType & value, double priority) {
    enqueueCount++;
    HeapEntry new_entry{
       value,
       priority,
       enqueueCount
    };

    heap.push_back(new_entry);

    int selecting_child = count;  //

    while (true) {
        int selecting_parent = (selecting_child - 1) / 2;

        if (selecting_child <= 0) { break; }
        if (takesPriority(selecting_parent, selecting_child)) { break; }

        swapHeapEntries(selecting_child, selecting_parent);
        selecting_child = selecting_parent;
    }

    count++;
}

template <typename ValueType>
ValueType PriorityQueue<ValueType>::dequeueHeap(){
    ValueType deleted_value = heap[0].value;

    swapHeapEntries(0, count - 1);
    heap.pop_back();
    count--;

    int selecting_parent = 0;
    int selecting_child = -1;

    while (true) {
        int left_child = 2 * selecting_parent + 1;
        int right_child = 2 * selecting_parent + 2;

        if (left_child >= count) { break; }
        else if (right_child >= count) { selecting_child = left_child; }
        else {
            if (takesPriority(left_child, right_child)) { selecting_child = left_child; }
            else { selecting_child = right_child; }
        }

        if (takesPriority(selecting_child, selecting_parent)) {
            swapHeapEntries(selecting_parent, selecting_child);
            selecting_parent = selecting_child;
            // selecting_child will be reassigned in the next loop
        }
        else { break; }
    }

    return deleted_value;
}

template <typename ValueType>
bool PriorityQueue<ValueType>::takesPriority(int i1, int i2) {
   /* Return true if i1 takes the priority */

    return (heap[i1].priority < heap[i2].priority) || ((heap[i1].priority == heap[i2].priority) && (heap[i1].sequence < heap[i2].sequence));
}

template <typename ValueType>
void PriorityQueue<ValueType>::swapHeapEntries(int i1, int i2) {
   /* Swap elements in the two positions */

    HeapEntry temp_entry = heap[i1];
    heap[i1] = heap[i2];
    heap[i2] = temp_entry;


}

template <typename ValueType>
std::string PriorityQueue<ValueType>::toString() {
   /* convert the PriorityQueue into a printable String */

    ostringstream temp_oss;
    PriorityQueue<ValueType> temp_pq = *this;

    temp_oss << "{";

    bool is_first_term = true;
    while (!temp_pq.isEmpty()) {
        if (is_first_term) { is_first_term = false; }
        else { temp_oss << ", "; }
        temp_oss << temp_pq.peekPriority() << ":\"" << temp_pq.peek() << "\"";

        temp_pq.dequeue();
    }

    temp_oss << "}";

    return temp_oss.str();
}
